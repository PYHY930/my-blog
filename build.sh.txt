#!/usr/bin/env bash
set -euo pipefail

# 1. 指定你要使用的 Hugo 版本（和你本地一致）
HUGO_VERSION=0.168.0

# 2. 从 GitHub 下载对应版本的 Hugo（Cloudflare 是 Linux 环境，所以下载 linux 版）
curl -sfL --output-dir "${PWD}" -O "https://github.com/gohugoio/hugo/releases/download/v${HUGO_VERSION}/hugo_${HUGO_VERSION}_linux-amd64.tar.gz"

# 3. 解压下载的文件
tar -xzf "hugo_${HUGO_VERSION}_linux-amd64.tar.gz"

# 4. 使用刚刚解压出来的最新版 Hugo 来生成静态网页
./hugo --gc --minify
